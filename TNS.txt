import numpy as np
import scipy.signal as signal
import soundfile as sf
import matplotlib.pyplot as plt


# Charger un fichier audio
file_path = "sp24_street_sn15.wav"  # Remplace par ton fichier audio
signal_audio, fs = sf.read(file_path)

# Vérifier si le signal est stéréo et le convertir en mono si nécessaire
if len(signal_audio.shape) > 1:
    signal_audio = np.mean(signal_audio, axis=1)

# Ajouter du bruit blanc
noise_level = 0.01
noise = noise_level * np.random.randn(len(signal_audio))
signal_noisy = signal_audio + noise

# Paramètres des filtres
wc = 3000 # Fréquence de coupure en Hz
order =8 # Ordre des filtres
rp = 0.5  # Ondulation en dB pour Chebyshev I et Cauer (Elliptique)
rs = 40  # Atténuation dans la bande d'arrêt (pour Cauer)

# Fréquence normalisée
nyquist = 0.5 * fs
normal_wc = wc / nyquist

# Création des filtres IIR
filters = {
    "Butterworth": signal.butter(order, normal_wc, btype='low', analog=False, output='ba'),
    "Chebyshev I": signal.cheby1(order, rp, normal_wc, btype='low', analog=False, output='ba'),
    "Cauer (Elliptique)": signal.ellip(order, rp, rs, normal_wc, btype='low', analog=False, output='ba')
}

# Appliquer les filtres et stocker les résultats
filtered_signals = {}
for name, (b, a) in filters.items():
    filtered_signal = signal.lfilter(b, a, signal_noisy)
    filtered_signals[name] = filtered_signal

# Fonction pour calculer le SNR
def snr(signal_clean, signal_noisy):
    power_signal = np.mean(signal_clean ** 2)
    power_noise = np.mean((signal_noisy - signal_clean) ** 2)
    return 10 * np.log10(power_signal / power_noise)

# Afficher le SNR avant et après filtrage
print("\nSNR avant filtrage :", snr(signal_audio, signal_noisy), "dB")
for name, filtered_signal in filtered_signals.items():
    print(f"SNR après filtrage ({name}): {snr(signal_audio, filtered_signal):.2f} dB")

# Sauvegarde des fichiers audio
sf.write("output_noisy.wav", signal_noisy, fs)
for name, filtered_signal in filtered_signals.items():
    sf.write(f"output_{name.replace(' ', '_').lower()}.wav", filtered_signal, fs)

# Tracer les réponses fréquentielles des filtres
plt.figure(figsize=(10, 6))
for name, (b, a) in filters.items():
    w, h = signal.freqz(b, a, worN=8000)
    plt.plot(w * fs / (2 * np.pi), 20 * np.log10(abs(h)), label=name)

plt.title("Réponse fréquentielle des filtres IIR")
plt.xlabel("Fréquence (Hz)")
plt.ylabel("Amplitude (dB)")
plt.legend()
plt.grid()
plt.show()

from scipy.fft import fft, fftfreq

def plot_spectrum(signal, fs, title):
    N = len(signal)
    freqs = fftfreq(N, 1/fs)
    spectrum = np.abs(fft(signal))

    plt.plot(freqs[:N//2], spectrum[:N//2])
    plt.title(title)
    plt.xlabel("Fréquence (Hz)")
    plt.ylabel("Amplitude")
    plt.show()

plot_spectrum(signal_noisy, fs, "Spectre du signal bruité")
plot_spectrum(filtered_signal, fs, "Spectre du signal filtré")

# Tracer les signaux pour chaque filtre
for name, filtered_signal in filtered_signals.items():
    plt.figure(figsize=(12, 8))

    plt.subplot(3, 1, 1)
    plt.plot(signal_audio, label="Signal original")
    plt.legend()
    plt.title("Signal original")

    plt.subplot(3, 1, 2)
    plt.plot(signal_noisy, label="Signal avec bruit", color='r')
    plt.legend()
    plt.title("Signal bruité")

    plt.subplot(3, 1, 3)
    plt.plot(filtered_signal, label=f"Signal filtré ({name})")
    plt.legend()
    plt.title(f"Signal après filtrage - {name}")

    plt.xlabel("Échantillons")
    plt.ylabel("Amplitude")
    plt.suptitle(f"Comparaison du signal original, bruité et filtré ({name})")
    plt.show()
